package egg

import (
	"context"
	"encoding/json"
	"os"
	"path/filepath"
	"testing"
	"time"
)

// Exercise the browser's actual three-second receipt wait, rather than a
// twenty-second wait that can hide the browser's late expiry clock.
func TestSessionAnswerThreeSecondDiagnostic(t *testing.T) {
	dir, home, cwd, _ := lifecycleFixture(t)
	id := filepath.Base(dir)
	lifecycleHook(t, home, id, "seq.00000000000000000001", `{"session_id":"ours","hook_event_name":"PermissionRequest","tool_name":"Bash"}`)
	initial := lifecycleRead(t, dir, home, cwd, 0, 100)
	cursor := initial.StateCursor
	sends := 0
	var enqueued time.Time
	o := SessionPromptOptions{RequestID: "diagnostic-answer", Input: "Yes", AnswerToStateCursor: &cursor, Timeout: 3 * time.Second,
		Read: func(ctx context.Context, after int64, limit int) (SessionView, error) {
			return ReadSessionLifecycle(dir, "claude", cwd, home, "ours", true, after, limit)
		},
		Send: func(context.Context, string) (PromptDelivery, error) {
			sends++
			enqueued = time.Now()
			return PromptDelivery{BytesEnqueued: 4}, nil
		},
	}
	receipt, err := SubmitSessionPrompt(t.Context(), dir, o)
	if err != nil {
		t.Fatal(err)
	}
	responseDelay := time.Since(enqueued).Milliseconds()
	if responseDelay < 2900 || responseDelay > 10000 || sends != 1 || receipt.Status != "unconfirmed" || !receipt.TransportEnqueued || receipt.NativeReceiptObserved {
		t.Fatalf("delay=%dms sends=%d receipt=%+v", responseDelay, sends, receipt)
	}
	// This is native tool activity, not a fabricated user-message receipt.
	lifecycleHook(t, home, id, "seq.00000000000000000002", `{"session_id":"ours","hook_event_name":"PreToolUse","tool_name":"Bash"}`)
	working := lifecycleRead(t, dir, home, cwd, 0, 100)
	for _, event := range working.Events {
		if event.Type == "message" || event.Type == "prompt_submitted" {
			t.Fatal("unexpected native user receipt")
		}
	}
	replay, err := SubmitSessionPrompt(t.Context(), dir, o)
	if err != nil || sends != 1 || replay.Status != "waiting_answer_delivered" || !replay.Retried {
		t.Fatalf("replay=%+v sends=%d err=%v", replay, sends, err)
	}
	data := map[string]any{"response_delay_ms": responseDelay, "receipt": receipt, "replay": replay, "working": working, "sends": sends}
	encoded, err := json.MarshalIndent(data, "", "  ")
	if err != nil {
		t.Fatal(err)
	}
	if err = os.WriteFile(os.Getenv("SESSION_ANSWER_OBSERVATION"), append(encoded, '\n'), 0600); err != nil {
		t.Fatal(err)
	}
	t.Logf("real receipt wait=%dms; status=%s; replay=%s; actual sends=%d", responseDelay, receipt.Status, replay.Status, sends)
}
